class Solution {
    public int[] getSumAbsoluteDifferences(int[] nums) {
        int n = nums.length;
        int[] result = new int[n];

        int totalSum = 0;

        // Find total sum
        for (int num : nums) {
            totalSum += num;
        }

        int leftSum = 0;

        for (int i = 0; i < n; i++) {
            int rightSum = totalSum - leftSum - nums[i];

            // Difference with elements on the left
            int leftCost = nums[i] * i - leftSum;

            // Difference with elements on the right
            int rightCount = n - i - 1;
            int rightCost = rightSum - nums[i] * rightCount;

            result[i] = leftCost + rightCost;

            leftSum += nums[i];
        }

        return result;
    }
}
